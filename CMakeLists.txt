cmake_minimum_required(VERSION 3.16)
project(yuca VERSION 0.1.1 LANGUAGES CXX)

include(CTest)
include(GNUInstallDirs)
option(YUCA_BUILD_DEMOS "Build shared/static library demos" ON)
option(YUCA_BUILD_BENCHMARKS "Build deterministic microbenchmarks" OFF)
option(YUCA_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(YUCA_BUILD_FUZZERS "Build Clang libFuzzer targets (also enables sanitizers)" OFF)
option(YUCA_ENABLE_CLANG_TIDY "Run the blocking clang-tidy checks on the core library" OFF)
option(YUCA_WARNINGS_AS_ERRORS "Treat first-party core compiler warnings as errors" OFF)
option(YUCA_ENABLE_COVERAGE "Instrument a separate Clang source/branch coverage build" OFF)
option(YUCA_BUILD_PROFILING "Build native CPU workloads and diagnostic allocation metrics" OFF)

if(YUCA_BUILD_PROFILING AND (YUCA_ENABLE_COVERAGE OR YUCA_ENABLE_SANITIZERS OR YUCA_BUILD_FUZZERS))
    message(FATAL_ERROR "Use a separate profiling build; diagnostic allocators must not replace sanitizer allocators")
endif()

if(YUCA_ENABLE_COVERAGE)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "Source coverage requires Clang")
    endif()
    if(YUCA_ENABLE_SANITIZERS OR YUCA_BUILD_FUZZERS OR YUCA_BUILD_BENCHMARKS)
        message(FATAL_ERROR "Use separate coverage, sanitizer/fuzzer, and benchmark builds")
    endif()
    if(NOT BUILD_TESTING)
        message(FATAL_ERROR "Coverage requires BUILD_TESTING=ON")
    endif()
endif()

if(YUCA_ENABLE_CLANG_TIDY)
    find_program(YUCA_CLANG_TIDY_EXECUTABLE NAMES clang-tidy clang-tidy-18 clang-tidy-19 clang-tidy-20)
    if(NOT YUCA_CLANG_TIDY_EXECUTABLE)
        message(FATAL_ERROR "clang-tidy not found; set YUCA_CLANG_TIDY_EXECUTABLE explicitly")
    endif()
endif()

if(YUCA_BUILD_FUZZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "libFuzzer targets require Clang")
    endif()
    set(YUCA_ENABLE_SANITIZERS ON)
endif()
if(YUCA_ENABLE_SANITIZERS AND NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizer configuration requires Clang or GCC")
endif()

function(yuca_configure_target target)
    set_target_properties(${target} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wparentheses -Wvla)
    endif()
    if(YUCA_ENABLE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
    if(YUCA_BUILD_FUZZERS)
        target_compile_options(${target} PRIVATE -fsanitize=fuzzer-no-link)
    endif()
    if(YUCA_ENABLE_COVERAGE)
        target_compile_options(${target} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
        target_link_options(${target} PRIVATE -fprofile-instr-generate)
    endif()
endfunction()

set(SOURCE_FILES src/yuca/key.cpp src/yuca/document.cpp src/yuca/indexer.cpp)
foreach(kind shared static)
    if(kind STREQUAL "shared")
        add_library(yuca_${kind} SHARED ${SOURCE_FILES})
    else()
        add_library(yuca_${kind} STATIC ${SOURCE_FILES})
    endif()
    add_library(yuca::${kind} ALIAS yuca_${kind})
    target_include_directories(yuca_${kind} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_compile_features(yuca_${kind} PUBLIC cxx_std_11)
    set_target_properties(yuca_${kind} PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
    yuca_configure_target(yuca_${kind})
    if(NOT MSVC)
        target_compile_options(yuca_${kind} PRIVATE -Wconversion -Wsign-conversion -Wshadow)
    endif()
    if(YUCA_WARNINGS_AS_ERRORS)
        if(MSVC)
            target_compile_options(yuca_${kind} PRIVATE /WX)
        else()
            target_compile_options(yuca_${kind} PRIVATE -Werror)
        endif()
    endif()
endforeach()

if(YUCA_ENABLE_CLANG_TIDY)
    # Static/shared compile identical core sources; analyze once, not twice.
    set_target_properties(yuca_static PROPERTIES CXX_CLANG_TIDY "${YUCA_CLANG_TIDY_EXECUTABLE}")
endif()

if(YUCA_BUILD_DEMOS)
    foreach(kind shared static)
        add_executable(yuca_demo_${kind} demo.cpp)
        target_link_libraries(yuca_demo_${kind} PRIVATE yuca_${kind})
        yuca_configure_target(yuca_demo_${kind})
        if(BUILD_TESTING)
            add_test(NAME demo_${kind} COMMAND yuca_demo_${kind})
        endif()
    endforeach()
endif()

if(BUILD_TESTING)
    add_executable(yuca_tests tests/utils_tests.cpp tests/document_tests.cpp
        tests/indexer_tests.cpp tests/regression_tests.cpp tests/tests_main.cpp)
    target_link_libraries(yuca_tests PRIVATE yuca_static)
    yuca_configure_target(yuca_tests)
    add_test(NAME unit_tests COMMAND yuca_tests)
    set_tests_properties(unit_tests PROPERTIES TIMEOUT 60)
    if(YUCA_ENABLE_CLANG_TIDY)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME clang_tidy_gate COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/clang_tidy_gate.py"
            "${YUCA_CLANG_TIDY_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/.clang-tidy")
        set_tests_properties(clang_tidy_gate PROPERTIES TIMEOUT 60)
    endif()
    if(UNIX)
        add_executable(yuca_catch_portability tests/catch_portability.cpp)
        yuca_configure_target(yuca_catch_portability)
        add_test(NAME catch_portability COMMAND yuca_catch_portability)
    endif()
    add_executable(yuca_manual_example docs/examples/basic.cpp)
    target_link_libraries(yuca_manual_example PRIVATE yuca_static)
    yuca_configure_target(yuca_manual_example)
    add_test(NAME manual_example COMMAND yuca_manual_example)
endif()

if(YUCA_BUILD_BENCHMARKS)
    add_executable(yuca_benchmarks benchmarks/benchmarks.cpp)
    target_link_libraries(yuca_benchmarks PRIVATE yuca_static)
    yuca_configure_target(yuca_benchmarks)
    if(BUILD_TESTING)
        add_test(NAME benchmark_smoke COMMAND yuca_benchmarks 1 50)
    endif()
endif()

if(YUCA_BUILD_PROFILING)
    add_executable(yuca_workload benchmarks/profile_workload.cpp)
    target_link_libraries(yuca_workload PRIVATE yuca_static)
    yuca_configure_target(yuca_workload)
    add_executable(yuca_allocations benchmarks/profile_workload.cpp tests/support/allocation_tracker.cpp)
    target_compile_definitions(yuca_allocations PRIVATE YUCA_ALLOCATION_METRICS=1)
    target_link_libraries(yuca_allocations PRIVATE yuca_static)
    yuca_configure_target(yuca_allocations)
    if(BUILD_TESTING)
        add_executable(yuca_allocation_tracker_tests tests/allocation_tracker_tests.cpp tests/support/allocation_tracker.cpp)
        yuca_configure_target(yuca_allocation_tracker_tests)
        add_test(NAME allocation_tracker COMMAND yuca_allocation_tracker_tests)
        foreach(workload index search search_grouped search_top10 search_distance)
            add_test(NAME profile_${workload} COMMAND yuca_workload ${workload} 24 2)
            add_test(NAME allocations_${workload} COMMAND yuca_allocations ${workload} 24 2)
        endforeach()
    endif()
endif()

if(YUCA_BUILD_FUZZERS)
    foreach(name utils indexer)
        add_executable(yuca_fuzz_${name} fuzz/${name}_fuzz.cpp)
        target_link_libraries(yuca_fuzz_${name} PRIVATE yuca_static)
        yuca_configure_target(yuca_fuzz_${name})
        target_link_options(yuca_fuzz_${name} PRIVATE -fsanitize=fuzzer)
        if(BUILD_TESTING)
            add_test(NAME fuzz_${name}_corpus COMMAND yuca_fuzz_${name}
                -runs=0 -max_len=512 "${CMAKE_CURRENT_SOURCE_DIR}/fuzz/corpus/${name}")
            set_tests_properties(fuzz_${name}_corpus PROPERTIES TIMEOUT 30
                ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1")
        endif()
    endforeach()
endif()

if(YUCA_ENABLE_COVERAGE)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    get_filename_component(YUCA_COMPILER_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
    find_program(YUCA_LLVM_COV NAMES llvm-cov llvm-cov-18 llvm-cov-19 llvm-cov-20 HINTS "${YUCA_COMPILER_DIRECTORY}")
    find_program(YUCA_LLVM_PROFDATA NAMES llvm-profdata llvm-profdata-18 llvm-profdata-19 llvm-profdata-20 HINTS "${YUCA_COMPILER_DIRECTORY}")
    if(NOT YUCA_LLVM_COV OR NOT YUCA_LLVM_PROFDATA)
        message(FATAL_ERROR "Set YUCA_LLVM_COV and YUCA_LLVM_PROFDATA to tools matching this Clang")
    endif()
    add_test(NAME coverage_report_tests COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_coverage_report.py")
    add_custom_target(coverage
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/coverage_report.py"
            --build-dir "${CMAKE_BINARY_DIR}" --source-dir "${CMAKE_CURRENT_SOURCE_DIR}"
            --llvm-cov "${YUCA_LLVM_COV}" --llvm-profdata "${YUCA_LLVM_PROFDATA}"
        DEPENDS yuca_tests yuca_manual_example
        USES_TERMINAL VERBATIM)
    foreach(test_target yuca_demo_shared yuca_demo_static yuca_catch_portability)
        if(TARGET ${test_target})
            add_dependencies(coverage ${test_target})
        endif()
    endforeach()
endif()

install(TARGETS yuca_shared yuca_static
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY src/yuca DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.hpp")
